🔗 Step-by-Step: Connecting Python with SQLite

  1. Import sqlite3 module
    import sqlite3
  2. Connect to a database (or create it)
    conn = sqlite3.connect('school.db')
  3. Create a cursor object
    cur = conn.cursor()
  4. Create a table 
    cur.execute('''
CREATE TABLE IF NOT EXISTS student (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    grade INTEGER NOT NULL
)
''')
  5. Insert data 
    cur.execute("INSERT INTO student (name, grade) VALUES (?, ?)", ("Aman", 95))
cur.execute("INSERT INTO student (name, grade) VALUES (?, ?)", ("Preet", 88))
conn.commit()
  6. Read data 
    cur.execute("SELECT * FROM student")
rows = cur.fetchall()
for row in rows:
    print(row)
  7. Update data 
    cur.execute("UPDATE student SET grade = 92 WHERE name = 'Preet'")
conn.commit()
  8. Delete data 
    cur.execute("DELETE FROM student WHERE name = 'Aman'")
conn.commit()
  9. Filter data (WHERE clause) 
    cur.execute("SELECT * FROM student WHERE grade > 90")
print(cur.fetchall())
  10. Sort data (ORDER BY) 
    cur.execute("SELECT * FROM student ORDER BY grade DESC")
print(cur.fetchall())
  11. Count total rows 
    cur.execute("SELECT COUNT(*) FROM student")
print("Total students:", cur.fetchone()[0])
  12. Close the connection
    conn.close()

📦 Full Example

import sqlite3
conn = sqlite3.connect('school.db')
cur = conn.cursor()

# Create table
cur.execute('''
CREATE TABLE IF NOT EXISTS student (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    grade INTEGER NOT NULL
)
''')

# Insert
cur.execute("INSERT INTO student (name, grade) VALUES (?, ?)", ("Preet", 88))
cur.execute("INSERT INTO student (name, grade) VALUES (?, ?)", ("Aman", 95))
conn.commit()

# Update
cur.execute("UPDATE student SET grade = 92 WHERE name = 'Preet'")
conn.commit()

# Delete
cur.execute("DELETE FROM student WHERE name = 'Aman'")
conn.commit()

# Read
cur.execute("SELECT * FROM student")
print("All Students:", cur.fetchall())

# Filter
cur.execute("SELECT * FROM student WHERE grade > 90")
print("High scorers:", cur.fetchall())

# Sort
cur.execute("SELECT * FROM student ORDER BY grade DESC")
print("Sorted:", cur.fetchall())

# Count
cur.execute("SELECT COUNT(*) FROM student")
print("Total students:", cur.fetchone()[0])

# Close
conn.close()

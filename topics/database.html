🔗 Step-by-Step: Connecting Python with SQLite

  1. Import sqlite3 module
    import sqlite3
  2. Connect to a database (or create it)
    conn = sqlite3.connect('school.db')
  3. Create a cursor object
    cur = conn.cursor()
  4. Create a table 
    cur.execute('''
CREATE TABLE IF NOT EXISTS student (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    grade INTEGER NOT NULL
)
''')
  5. Insert data 
    cur.execute("INSERT INTO student (name, grade) VALUES (?, ?)", ("Aman", 95))
cur.execute("INSERT INTO student (name, grade) VALUES (?, ?)", ("Preet", 88))
conn.commit()
  6. Read data 
    cur.execute("SELECT * FROM student")
rows = cur.fetchall()
for row in rows:
    print(row)
  7. Update data 
    cur.execute("UPDATE student SET grade = 92 WHERE name = 'Preet'")
conn.commit()
  8. Delete data 
    cur.execute("DELETE FROM student WHERE name = 'Aman'")
conn.commit()
  9. Filter data (WHERE clause) 
    cur.execute("SELECT * FROM student WHERE grade > 90")
print(cur.fetchall())
  10. Sort data (ORDER BY) 
    cur.execute("SELECT * FROM student ORDER BY grade DESC")
print(cur.fetchall())
  11. Count total rows 
    cur.execute("SELECT COUNT(*) FROM student")
print("Total students:", cur.fetchone()[0])
  12. Close the connection
    conn.close()

📦 Full Example

import sqlite3
conn = sqlite3.connect('school.db')
cur = conn.cursor()

# Create table
cur.execute('''
CREATE TABLE IF NOT EXISTS student (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    grade INTEGER NOT NULL
)
''')

# Insert
cur.execute("INSERT INTO student (name, grade) VALUES (?, ?)", ("Preet", 88))
cur.execute("INSERT INTO student (name, grade) VALUES (?, ?)", ("Aman", 95))
conn.commit()

# Update
cur.execute("UPDATE student SET grade = 92 WHERE name = 'Preet'")
conn.commit()

# Delete
cur.execute("DELETE FROM student WHERE name = 'Aman'")
conn.commit()

# Read
cur.execute("SELECT * FROM student")
print("All Students:", cur.fetchall())

# Filter
cur.execute("SELECT * FROM student WHERE grade > 90")
print("High scorers:", cur.fetchall())

# Sort
cur.execute("SELECT * FROM student ORDER BY grade DESC")
print("Sorted:", cur.fetchall())

# Count
cur.execute("SELECT COUNT(*) FROM student")
print("Total students:", cur.fetchone()[0])

# Close
conn.close()

🛡️ Why ? Instead of an f-string — SQL Injection

Every insert above used VALUES (?, ?) plus a separate tuple of values — that's not just style, it's a security requirement.

❌ The Dangerous Way

name = "Aman'); DROP TABLE student; --"
cur.execute(f"INSERT INTO student (name, grade) VALUES ('{name}', 90)")
# The attacker's input closes the string early and injects a NEW SQL command —
# this can delete your entire table.

✅ The Safe Way

name = "Aman'); DROP TABLE student; --"
cur.execute("INSERT INTO student (name, grade) VALUES (?, ?)", (name, 90))
# The ? placeholder treats "name" as plain data, never as SQL code —
# it gets inserted literally, harmlessly, as a weird-looking name.
🚨 Never build SQL with f-strings or + concatenation using outside input (form fields, API data, user typing). Always use ? placeholders — this attack is called SQL injection and is one of the most common real-world security bugs.

↩️ Transactions & Rollback

A transaction groups several changes together — either all of them save, or none do. Useful when one action depends on another succeeding (like transferring marks between two students).

import sqlite3

conn = sqlite3.connect('school.db')
cur = conn.cursor()

try:
    cur.execute("UPDATE student SET grade = grade - 10 WHERE name = 'Aman'")
    cur.execute("UPDATE student SET grade = grade + 10 WHERE name = 'Preet'")
    conn.commit()          # both updates saved together
    print("Transfer successful")
except sqlite3.Error as e:
    conn.rollback()         # undo BOTH updates if anything failed
    print(f"Transfer failed, rolled back: {e}")

If the second UPDATE fails (e.g. Preet doesn't exist), rollback() undoes the first one too — Aman doesn't lose marks for nothing.

🔗 Joins & Foreign Keys — Linking Tables

Real apps split data across multiple tables and connect them with a shared ID — a foreign key.

cur.execute('''
CREATE TABLE IF NOT EXISTS class (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    class_name TEXT NOT NULL
)
''')

cur.execute('''
CREATE TABLE IF NOT EXISTS student (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    grade INTEGER NOT NULL,
    class_id INTEGER,
    FOREIGN KEY (class_id) REFERENCES class(id)
)
''')

cur.execute("INSERT INTO class (class_name) VALUES ('10-A')")
cur.execute("INSERT INTO student (name, grade, class_id) VALUES (?, ?, ?)", ("Ravi", 88, 1))
conn.commit()

class_id in student points to the id in class — that link is the foreign key. Use JOIN to pull matching rows from both tables in one query:

cur.execute('''
SELECT student.name, student.grade, class.class_name
FROM student
JOIN class ON student.class_id = class.id
''')
print(cur.fetchall())
[('Ravi', 88, '10-A')]

⚠️ Handling Database Errors

import sqlite3

conn = sqlite3.connect('school.db')
cur = conn.cursor()

try:
    cur.execute("INSERT INTO student (name, grade) VALUES (?, ?)", (None, 90))
    conn.commit()
except sqlite3.IntegrityError as e:
    print(f"Data rule violated: {e}")   # e.g. NOT NULL constraint failed
except sqlite3.OperationalError as e:
    print(f"SQL/table problem: {e}")    # e.g. table doesn't exist, bad syntax
finally:
    conn.close()
Data rule violated: NOT NULL constraint failed: student.name
ExceptionCommon cause
IntegrityErrorViolated a rule like NOT NULL, UNIQUE, or a foreign key constraint
OperationalErrorBad SQL syntax, or table/column doesn't exist
